cmake_minimum_required(VERSION 3.20)
project(mpemtrans_profile_capture LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_path(CRYPTLIB_INCLUDE_DIR cryptlib.h REQUIRED)
set(GENERATOR_ARGUMENTS "${CRYPTLIB_INCLUDE_DIR}/cryptlib.h" "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  list(APPEND GENERATOR_ARGUMENTS --windows-x86)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_proxy.py" ${GENERATOR_ARGUMENTS}
                COMMAND_ERROR_IS_FATAL ANY)
add_library(capture SHARED capture_proxy.c)
target_include_directories(capture PRIVATE "${CRYPTLIB_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(capture PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_sources(capture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/proxy.def")
  target_compile_definitions(capture PRIVATE _CRT_SECURE_NO_WARNINGS)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set_target_properties(capture PROPERTIES OUTPUT_NAME cl32 PREFIX "")
  else()
    set_target_properties(capture PROPERTIES OUTPUT_NAME cl64 PREFIX "")
  endif()
else()
  set_target_properties(capture PROPERTIES OUTPUT_NAME cl PREFIX "lib")
endif()
